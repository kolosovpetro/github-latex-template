(* ================================================================ *)
(*  Definitions                                                     *)
(* ================================================================ *)

ClearAll[n,L, ForwardDifference];

(* --------------------------------------------------------------- *)
(*  Convention: x^0 = 1 for all x                                   *)
(* --------------------------------------------------------------- *)

Begin["`Private`"]

Unprotect[Power];
Power[_?NumericQ | _, 0] = 1;
Protect[Power];

End[];

(* --------------------------------------------------------------- *)
(*  Binomial coefficient appearing in the decomposition             *)
(* --------------------------------------------------------------- *)

L[n_, r_, t_, k_] :=
  Binomial[n - t + k/2 + r - 1, k + r]
  + Binomial[n - t + k/2 + r, k + r]
  - Sum[
      (Binomial[s - t + k/2, k + s + 1]
       + Binomial[s - t + k/2 + 1, k + s + 1])
      * Binomial[r - s + n - 2, r - s - 1],
      {s, 0, r - 1}
    ];

(* --------------------------------------------------------------- *)
(*  Central difference operator                                     *)
(* --------------------------------------------------------------- *)

CentralDifference[x_, n_, k_] :=
  Sum[
    (-1)^j
      Binomial[k, j]
      (x + k/2 - j)^n,
    {j, 0, k}
  ];


(* ================================================================ *)
(*  Parameters                                                      *)
(* ================================================================ *)

m = 3;
r = 2;


(* ================================================================ *)
(*  Compute coefficients and forward differences                    *)
(* ================================================================ *)

Do[

  Print["------------------------------------------------------------"];
  Print["m = ", m, ",   t = ", t];
  Print["------------------------------------------------------------"];

  Print["L[n,t,k]:"];
  Table[
    Together @ Expand @ L[n, r, t, k],
    {k, 0, m}
  ] // Print;

  Print["Central differences:"];
  Table[
    CentralDifference[t, m, k],
    {k, 0, m}
  ] // Print;

  Print[""],

  {t, 0, 4}

];
