(* ================================================================ *)
(*  Validates Proposition: Ordinary sums of powers                  *)
(* ================================================================ *)

ClearAll[
  KnuthRFoldSum,
  CentralDifference,
  ValidateOrdinaryCentralSumsOfPowers
];

(* --------------------------------------------------------------- *)
(*  Convention: x^0 = 1 for all x                                  *)
(* --------------------------------------------------------------- *)

Begin["`Private`"]

Unprotect[Power];
Power[_?NumericQ | _, 0] = 1;
Protect[Power];

End[];

(* --------------------------------------------------------------- *)
(*  Eulerian numbers (as given)                                    *)
(* --------------------------------------------------------------- *)

EulerianNumber[0, 0] = 1;
EulerianNumber[n_, k_] /; k < 0 || k >= n := 0;
EulerianNumber[n_, k_] :=
  EulerianNumber[n, k] =
    (k + 1) EulerianNumber[n - 1, k] +
    (n - k) EulerianNumber[n - 1, k - 1];

(* --------------------------------------------------------------- *)
(*  Multifold sums of powers                                       *)
(* --------------------------------------------------------------- *)

KnuthRFoldSum[0, n_, m_] := n^m;

KnuthRFoldSum[r_, n_, m_] :=
  Sum[KnuthRFoldSum[r - 1, k, m], {k, 1, n}];

(* --------------------------------------------------------------- *)
(*  Central finite difference                                      *)
(* --------------------------------------------------------------- *)

CentralDifference[t_, m_, k_] :=
  Sum[
    (-1)^j Binomial[k, j] (t + k/2 - j)^m,
    {j, 0, k}
  ];

(* --------------------------------------------------------------- *)
(*  Main identity                                                  *)
(* --------------------------------------------------------------- *)

OrdinaryCentralSumsOfPowers[n_, m_, t_] :=
  1/2 Sum[
    (
      Binomial[n - t + k/2, k + 1]
      + Binomial[n - t + k/2 + 1, k + 1]
      - Binomial[-t + k/2, k + 1]
      - Binomial[1 - t + k/2, k + 1]
    )
    * CentralDifference[t, m, k],
    {k, 0, m}
  ];

(* --------------------------------------------------------------- *)
(*  Validation routine                                             *)
(* --------------------------------------------------------------- *)

ValidateOrdinaryCentralSumsOfPowers[max_] :=
  Flatten[
    Table[
      FullSimplify[
        KnuthRFoldSum[1, n, m]
        -
        OrdinaryCentralSumsOfPowers[n, m, t]
      ],
      {n, 1, max},
      {m, 0, max},
      {t, -max, max}
    ]
  ];

(* --------------------------------------------------------------- *)
(*  Run test                                                       *)
(* --------------------------------------------------------------- *)

ValidateOrdinaryCentralSumsOfPowers[5]
