(*  Validates Proposition: Multifold Eulerian sums of powers       *)

ClearAll[
  KnuthRFoldSum,
  EulerianNumber,
  ValidateMultifoldEulerianSumsOfPowers
];

(* --------------------------------------------------------------- *)
(*  Convention: x^0 = 1 for all x                                  *)
(* --------------------------------------------------------------- *)

Begin["`Private`"]

Unprotect[Power];
Power[_?NumericQ | _, 0] = 1;
Protect[Power];

End[];

(* --------------------------------------------------------------- *)
(*  Eulerian numbers (as given)                                    *)
(* --------------------------------------------------------------- *)

EulerianNumber[0, 0] = 1;
EulerianNumber[n_, k_] /; k < 0 || k >= n := 0;
EulerianNumber[n_, k_] :=
  EulerianNumber[n, k] =
    (k + 1) EulerianNumber[n - 1, k] +
    (n - k) EulerianNumber[n - 1, k - 1];

(* --------------------------------------------------------------- *)
(*  Multifold sums of powers                                       *)
(* --------------------------------------------------------------- *)

KnuthRFoldSum[0, n_, m_] := n^m;

KnuthRFoldSum[r_, n_, m_] :=
  Sum[KnuthRFoldSum[r - 1, k, m], {k, 1, n}];

(* --------------------------------------------------------------- *)
(*  Main identity kernel                                            *)
(* --------------------------------------------------------------- *)

MultifoldEulerianSumsOfPowers[r_, n_, m_, t_] :=
  Sum[
    Sum[
      (
        Binomial[n - t + k + r - 1, k + r]
        - Sum[
            Binomial[k - t + r - 1 - s, k + r - s]*
              Binomial[s + n - 1, s],
            {s, 0, r - 1}
          ]
      )
      * Binomial[t + j - k, m - k]
      * EulerianNumber[m, j],
      {j, 0, m}
    ],
    {k, 0, m}
  ];

(* --------------------------------------------------------------- *)
(*  Validation routine                                              *)
(* --------------------------------------------------------------- *)

ValidateMultifoldEulerianSumsOfPowers[max_] :=
