\begin{theorem} [Multifold sums of powers]
  \label{thm:multifold-sum-of-powers}
  For non-negative integers $r,n,m$, and an arbitrary integer $t$
  \begin{align*}
    \KnuthRFoldSum{r}{n}{m}
    =
    \sum_{k=0}^{m}
    \left[
      \binom{n-t+k+r-1}{k+r}
      -
      \sum_{s=0}^{r-1}
      \binom{k-t+r-1-s}{k+r-s} \KnuthRFoldSum{s}{n}{0}
    \right] \nabla^{k} t^{m}.
  \end{align*}
\end{theorem}
